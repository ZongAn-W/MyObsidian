- 将主串中模式串相同长度的子串依次对比，当子串与模式串某个对应字符不匹配时，就放弃当前子串，继而检索下一个子串，直到找到一个完全匹配的子串或全不匹配。

```c
int Index(SString S, SString T) {
    int k = 1;  // 表示初始化比较的是主串中的第一个子串
    int i = k, j = 1;
    while (i <= S.length && j <= T.length) {
        if (S.ch[i] == T.ch[j]) {
            ++i; ++j;  // 继续比较后继字符
        } else {
            k++;  // 检查下一个子串
            i = k; j = 1;  // 指针后退重新开始匹配
        }
    }
    if (j > T.length) return k;
    else return 0;
}
```

```cpp
int Index(SString S, SString T) {
    int i = 1, j = 1;
    while (i <= S.length && j <= T.length) {
        if (S.ch[i] == T.ch[j]) {
            ++i; ++j;  // 继续比较后继字符
        } else {
            i = i - j + 2;
            j = 1;  // 指针后退重新开始匹配
        }
    }
    if (j > T.length) return i - T.length;
    else return 0;
}
