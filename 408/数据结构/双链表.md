- 结点中有两个指针prior和next，分别指向其前驱结点和后继结点
```c
typedef struct DNode{ // 定义双链表结点类型
    ElemType data; // 数据域
    struct DNode *prior; // 直接前驱
    struct DNode *next; // 直接后继
} DNode, *DLinkList;
```

### 初始化
```c
Status InitDLinkList(DLinklist &L){
    L=(DNode *)malloc(sizeof(DNode)); // 分配一个头结点
    if(L==NULL) return ERROR; // 内存不足，分配失败
    L->prior=NULL; // 头结点的 prior 永远指向 NULL
    L->next=NULL; // 头结点之后暂时还没有结点
    return OK;
}
```
### 插入
```c
Status Listinsert_DL(DLinkList &L, int i, ElemType e){
// 在带头结点的双向链表 L 中第 i 个位置之后插入元素 e
    if(!(p=GetElem_DL(L, i))) // 在 L 中确定第 i 个元素的位置指针 p
        return ERROR; // p 为 NULL 时，第 i 个元素不存在
    s=new DNode; // 生成新结点 *s
    s->data=e; // 将结点 *s 数据域置为 e
    s->next=p->next;
    p->next->prior=s;
    s->prior=p;
    p->next=s;
    return OK;
}
```
- 如果是在最后一个结点后插入需要添加判断
### 删除
```c
Status Listdelete_DL(DLinkList &L,int i){
	if (!(p=GetElem_DL(L,i-1)))
		return ERROR;
	DNode *q = p->next;
	p->next = q->next;
	if (q->next != NULL)
		q->next->prior=p;
	free(q);
	return OK;
}
```
