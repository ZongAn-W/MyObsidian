- **初始化**

```cpp
// 初始化队列（带头结点）
void InitQueue(LinkQueue &Q) { // 构造一个空队列 Q
    Q.front = Q.rear = new LinkNode; // 生成新结点作为头结点，队头和队尾指针指向此结点
    Q.front->next = NULL; // 头结点的指针域置空
}

// 初始化队列（不带头结点）
void InitQueue(LinkQueue &Q) { // 构造一个空队列 Q
    Q.front = NULL;
    Q.rear = NULL;
}
``` 

- **入队：**
- 和循环队列的入队操作不同的是，链队在入队前不需要判断队是否满，需要为入队元素动态分配一个结点空间。

```cpp
// 带头结点的入队操作
Status EnQueue(LinkQueue &Q, ElemType e) { // 插入元素 e 为 Q 的新的队尾元素
    LinkNode *p = (LinkNode *)malloc(sizeof(LinkNode)); // 为入队元素分配结点空间，用指针 p 指向
    p->data = e; // 将新结点数据域置为 e
    p->next = NULL;
    Q.rear->next = p; // 将新结点插入到队尾
    Q.rear = p; // 修改队尾指针
    return OK;
}
// 不带头结点的入队操作
Status EnQueue(LinkQueue &Q, ElemType e) { // 插入元素 e 为 Q 的新的队尾元素
    LinkNode *p = new LinkNode; // 为入队元素分配结点空间，用指针 p 指向
    p->data = e; // 将新结点数据域置为 e
    p->next = NULL;
    if (Q.front == NULL) {
        Q.front = p;
        Q.rear = p; // 修改队头和队尾指针
    } else {
        Q.rear->next = p; // 将新结点插入到队尾
        Q.rear = p; // 修改队尾指针
    }
    return OK;
}
```

- **出队：**
- 和循环队列一样，链队在出队前也需要判断队列是否为空，不同的是，链队在出队后需要释放出队头元素的所占空间。

```cpp
// 带头结点的出队操作
Status DeQueue(LinkQueue &Q, ElemType &e) { // 删除 Q 的队头元素，用 e 返回其值
    if (Q.front == Q.rear)
        return ERROR; // 若队列空，则返回 ERROR
    LinkNode *p = Q.front->next; // p 指向队头元素
    e = p->data; // e 保存队头元素的值
    Q.front->next = p->next; // 修改头指针
    if (Q.rear == p)
        Q.rear = Q.front; // 最后一个元素被删，队尾指针指向头结点
    free(p); // 释放原队头元素的空间
    return OK;
}

// 不带头结点的出队操作
Status DeQueue(LinkQueue &Q, ElemType &e) {
    if (Q.front == NULL)
        return ERROR; // 空队
    LinkNode *p = Q.front; // p 指向此次出队的结点
    e = p->data; // e 返回队头元素
    Q.front = p->next; // 修改 front 指针
    if (Q.rear == p) { // 此次是最后一个结点出队
        Q.front = NULL;
        Q.rear = NULL;
    }
    free(p);
    return OK;
}
```
