- 将普通的二叉链表中的空指针改为指向前驱或后继的线索
- 可以设置两个指针通过一次遍历来获取并记录信息
【算法描述】设立两指针，一前一后，访问到某节点之后：
1. 如果其左指针为空，如果其左指针为空，则填充上此节点的前驱节点的地址，通过前指针获取
2. 如果前指针指向节点的右指针为空，则填充上当前访问节点的地址

```cpp
ThreadNode *pre = NULL; // 定义指针指向当前访问结点的前驱

void InThread(ThreadTree T) {
    if (T != NULL) {
        InThread(T->lchild); // 中序遍历左子树
        visit(T); // 访问根结点
        InThread(T->rchild); // 中序遍历右子树
    }
}

void visit(ThreadNode *p) {
    if (p->lchild == NULL) { // 左子树为空，建立前驱线索
        p->lchild = pre;
        p->ltag = 1;
    }
    if (pre != NULL && pre->rchild == NULL) {
        pre->rchild = p; // 建立前驱结点的后继线索
        pre->rtag = 1;
    }
    pre = p; // 标记当前结点成为刚刚访问过的结点
}
```

- 上述代码逻辑会导致当p指针遍历到最后一个结点时，并没有尝试填充其rchild域

```cpp
// 中序遍历对二叉树线索化
void InThread(ThreadTree &p, ThreadTree &pre) {
    if (p != NULL) {
        InThread(p->lchild, pre);
        if (p->lchild == NULL) {
            p->lchild = pre;
            p->ltag = 1;
        }
        if (pre != NULL && pre->rchild == NULL) {
            pre->rchild = p;
            pre->rtag = 1;
        }
        pre = p;
        InThread(p->rchild, pre);
    }
}

// 中序遍历建立中序线索二叉树
void CreateInThread(ThreadTree T) {
    ThreadTree pre = NULL;
    if (T != NULL) { // 非空树
        InThread(T, pre);
        if (pre->rchild == NULL)
            pre->rtag = 1;
    }
}
```

- **在中序线索二叉树中找中序后继**

（1）若 `p->rtag == 1`，则表明其后继 `next = p->rchild`。  
（2）若 `p->rtag == 0`，表示其右指针指向右孩子，并不是中序后继。
在中序线索二叉树中找指定结点 `p` 的中序后继 `next`，若 `p->rtag` 为 `1`，则 `next` 为 `p->rchild`；若 `p->rtag` 为 `0`，则其后继为此结点右子树的最下层左结点。

```cpp
ThreadNode *Nextnode(ThreadNode *p) { // 找结点 p 在中序序列下的后继
    if (p->rtag == 0)
        return Firstnode(p->rchild);
    else
        return p->rchild; // rtag == 1，直接返回后继线索
}

// 找以 p 为根的子树中，最下层的左结点，即第一个被中序遍历的结点
ThreadNode *Firstnode(ThreadNode *p) {
    while (p->ltag == 0)
        p = p->lchild; // 最左下结点（不一定是叶结点）
    return p;
}

void Inorder(ThreadNode *T) { // 对中序线索二叉树进行中序遍历
    for (ThreadNode *p = Firstnode(T); p != NULL; p = Nextnode(p))
        visit(p);
}
```

- **在中序线索二叉树中找中序前驱**

```cpp
// 找结点 p 在中序序列下的前驱
ThreadNode *Prenode(ThreadNode *p) {
    if (p->ltag == 0)
        return Lastnode(p->lchild);
    else
        return p->lchild;
}

// 找以 p 为根的子树中，最后一个被中序遍历的结点
ThreadNode *Lastnode(ThreadNode *p) {
    while (p->rtag == 0)
        p = p->rchild;
    return p;
}

// 对中序线索二叉树进行逆向中序遍历
void RevInorder(ThreadNode *T) {
    for (ThreadNode *p = Lastnode(T); p != NULL; p = Prenode(p))
        visit(p);
}
```
