### 静态分配的初始化
• 初始化：静态分配
```c
#define MAXSIZE 100

typedef struct {
    ElemType data[MaxSize];
    int length;
} SqList;

void InitList(SqList &L){
    for(int i=0; i<MaxSize; i++){
        L.data[i]=0; // 将所有数据元素设置为默认初始值,这一步可有可无
    }
    L.length=0; // 顺序表初始长度为 0
}

int main(){
    SqList L; // 定义一个顺序表
    InitList(L); // 初始化顺序表
    ......
    return 0;
}
```

### 动态分配的初始化
• 初始化：动态分配
```c
typedef struct{
    ElemType *data; // 指向动态数组的指针
    int length; // 顺序表当前长度
    int MaxSize; // 顺序表最大容量
} SeqList;
```
- ==C 语言==中用 `malloc`、`free` 函数来实现动态申请和释放内存空间：  $L.data=(ElemType*)malloc(sizeof(ElemType)*InitSize);$
- 【注意】`malloc` 函数返回一个指针，需要强制转换类型为你定义的数据元素类型指针。  

- ==C++== 中可以用 `new` 和 `delete` 关键字来实现动态分配：  $L.data=\text{new ElemType }[InitSize];$  


• 初始化：动态分配
```c
#define InitSize 10 // 顺序表的初始长度

typedef struct{
    int *data; // 指向动态数组的指针
    int length; // 顺序表当前长度
    int MaxSize; // 顺序表最大容量
} SeqList;

void InitList(SeqList &L){
    // 用 malloc 函数申请一片连续的存储空间
    L.data=(int *)malloc(sizeof(int)*InitSize);
    L.length=0; // 空表长度为 0
    L.MaxSize=InitSize; // 初始存储容量
}

void IncreaseSize(SeqList &L,int len){ // 增加动态数组的长度
    int *p=L.data;
    L.data=(int *)malloc((L.MaxSize+len)*sizeof(int));
    for(int i=0; i<L.length; i++){
        L.data[i]=p[i]; // 将数据复制到新区域
    }
    L.MaxSize=L.MaxSize+len; // 顺序表最大长度增加 len
    free(p); // 释放原来的内存空间
}

int main(){
    SeqList L; // 定义一个顺序表
    InitList(L); // 初始化顺序表
    ... // 往顺序表中随便插入几个元素
    IncreaseSize(L,10);
    return 0;
}
```
