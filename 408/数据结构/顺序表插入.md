- 在表 L 的第 i 个位置上插入指定元素
```c
// 在 L 的位序 i 处插入元素 e
void ListInsert(SqList &L, int i, int e){
    for(int j=L.length; j>=i; j--) // 将第 i 个元素及之后的元素后移
        L.data[j]=L.data[j-1];
    L.data[i-1]=e; // 在位置 i 处放入 e
    L.length++; // 长度加 1
}
```

Status ListInsert(SqList &L, int i, int e){
    if(i<1 || i>L.length+1) return ERROR; // 判断 i 的范围是否有效
    if(L.length>=MaxSize) return ERROR; // 当前存储空间已满，不能插入
    for(int j=L.length; j>=i; j--) // 将第 i 个元素及之后的元素后移
        L.data[j]=L.data[j-1];
    L.data[i-1]=e; // 在位置 i 处放入 e
    L.length++; // 长度加 1
    return OK;
}
